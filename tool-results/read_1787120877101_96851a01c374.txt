    88→// ── Schema ─────────────────────────────────────────────────────────────

    90→// Strip non-digits, then validate. Used inside superRefine.
    91→function digits(s: string) {
    92→  return (s ?? "").replace(/\D/g, "");
    93→}
    94→
    95→const schema = z
    96→  .object({
    97→    // Contact
    98→    contactEmail: z.email("Enter a valid email address"),
    99→    contactPhone: z
   100→      .string()
   101→      .min(1, "Phone is required")
   102→      .refine(
   103→        (v) => /^(\+91[\-\s]?)?[6-9]\d{9}$/.test((v ?? "").trim()),
   104→        "Enter a 10-digit Indian mobile number"
   105→      ),
   106→    // Delivery address
   107→    fullName: z.string().min(2, "Please enter the recipient's name"),
   108→    line1: z.string().min(3, "Address line 1 is required"),
   109→    line2: z.string().optional(),
   110→    city: z.string().min(2, "City is required"),
   111→    state: z.string().min(2, "State is required"),
   112→    postcode: z
   113→      .string()
   114→      .min(1, "PIN code is required")
   115→      .refine((v) => /^\d{6}$/.test((v ?? "").trim()), "PIN code must be 6 digits"),
   116→    country: z.string().min(2, "Country is required"),
   117→    // Delivery method
   118→    deliveryMethod: z.enum(["standard", "express", "sameday"]),
   119→    // Payment
   120→    paymentMethod: z.enum(["card", "upi", "cod"]),
   121→    cardNumber: z.string().optional(),
   122→    cardExpiry: z.string().optional(),
   123→    cardCvv: z.string().optional(),
   124→    cardName: z.string().optional(),
   125→    upiId: z.string().optional(),
   126→  })
   127→  .superRefine((data, ctx) => {
   128→    if (data.paymentMethod === "card") {
   129→      const num = digits(data.cardNumber ?? "");
   130→      if (num.length < 13 || num.length > 19) {
   131→        ctx.addIssue({
   132→          code: "custom",
   133→          path: ["cardNumber"],
   134→          message: "Enter a valid 13–19 digit card number",
   135→        });
   136→      }
   137→      if (!/^(0[1-9]|1[0-2])\/([0-9]{2})$/.test((data.cardExpiry ?? "").trim())) {
   138→        ctx.addIssue({
   139→          code: "custom",
   140→          path: ["cardExpiry"],
   141→          message: "Use MM/YY format",
   142→        });
   143→      }
   144→      if (!/^\d{3,4}$/.test((data.cardCvv ?? "").trim())) {
   145→        ctx.addIssue({
   146→          code: "custom",
   147→          path: ["cardCvv"],
   148→          message: "CVV must be 3–4 digits",
   149→        });
   150→      }
   151→      if ((data.cardName ?? "").trim().length < 2) {
   152→        ctx.addIssue({
   153→          code: "custom",
   154→          path: ["cardName"],
   155→          message: "Name on card is required",
   156→        });
   157→      }
   158→    } else if (data.paymentMethod === "upi") {
   159→      if (!/^[\w.\-]{2,}@[a-zA-Z]{2,}$/.test((data.upiId ?? "").trim())) {
   160→        ctx.addIssue({
   161→          code: "custom",
   162→          path: ["upiId"],
   163→          message: "Enter a valid UPI ID (e.g. name@oksbi)",
   164→        });
   165→      }
   166→    }
   167→  });

   748→function buildPaymentLabel(values: FormValues): string {
   749→  if (values.paymentMethod === "card") {
   750→    const num = digits(values.cardNumber ?? "");
   751→    const last4 = num.slice(-4);
   752→    return last4 ? `Card ending ${last4}` : "Card payment";
   753→  }
   754→  if (values.paymentMethod === "upi") {
   755→    const id = (values.upiId ?? "").trim();
   756→    return id ? `UPI · ${id}` : "UPI";
   757→  }
   758→  return "Cash on delivery";
   759→}

   803→function Field({
   804→  id,
   805→  label,
   806→  error,
   807→  hint,
   808→  className,
   809→  register,
   810→  ...inputProps
   811→}: {
   812→  id: string;
   813→  label: string;
   814→  error?: string;
   815→  hint?: string;
   816→  className?: string;
   817→  register: ReturnType<ReturnType<typeof useForm<FormValues>>["register"]>;
   818→} & Omit<React.ComponentProps<typeof Input>, "id">) {
   819→  const describedBy = error
   820→    ? `${id}-error`
   821→    : hint
   822→    ? `${id}-hint`
   823→    : undefined;
   824→  return (
   825→    <div className={cn("space-y-1.5", className)}>
   826→      <Label htmlFor={id}>{label}</Label>
   827→      <Input
   828→        id={id}
   829→        aria-invalid={!!error}
   830→        aria-describedby={describedBy}
   831→        {...register}
   832→        {...inputProps}
   833→      />
   834→      {hint && !error && (
   835→        <p id={`${id}-hint`} className="text-xs text-muted-foreground">
   836→          {hint}
   837→        </p>
   838→      )}
   839→      {error && (
   840→        <p id={`${id}-error`} role="alert" className="text-sm text-destructive">
   841→          {error}
   842→        </p>
   843→      )}
   844→    </div>
   845→  );
